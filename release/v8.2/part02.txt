local function mapHasAny(names)
    local map = workspace:FindFirstChild("Map")
    if not map then return false end
    for _, name in ipairs(names) do
        if map:FindFirstChild(name, true) then return true end
    end
    return false
end

local function getCurrentSea()
    local placeId = tonumber(game.PlaceId)
    for sea, ids in pairs(SEA_PLACE_IDS) do
        if ids[placeId] then return sea end
    end

    -- Fallback for future Blox Fruits updates that introduce another PlaceId.
    -- Use several unique map markers instead of guessing Sea 3.
    if mapHasAny({"Castle on the Sea", "Castle On The Sea", "Hydra Island", "Hydra", "Floating Turtle", "Turtle"}) then
        return "Sea 3"
    end
    if mapHasAny({"Cafe", "Kingdom of Rose", "Dressrosa", "Green Zone", "Cursed Ship"}) then
        return "Sea 2"
    end
    if mapHasAny({"Middle Town", "Jungle", "Marine Fortress", "MarineFord", "Fountain City", "Skylands"}) then
        return "Sea 1"
    end

    return "Unknown"
end

local function getSafeZone()
    return SafeZones[getCurrentSea()]
end

local logCounter = 0
local logContainer = nil

local function AddLog(message, color)
    logCounter = logCounter + 1

    -- Every PrimeHub log line is mirrored to the executor console.
    -- This runs even before the GUI/log panel exists.
    executorLog("INFO", message)

    if typeof(color) ~= "Color3" then
        color = Color3.fromRGB(190, 205, 225)
    end

    if not logContainer or not logContainer.Parent then
        return
    end

    local label = Instance.new("TextLabel", logContainer)
    label.LayoutOrder = logCounter
    label.Size = UDim2.new(1, -4, 0, 14)
    label.BackgroundTransparency = 1
    label.Text = os.date("%H:%M:%S") .. " " .. tostring(message)
    label.TextColor3 = color
    label.Font = Enum.Font.Gotham
    label.TextSize = 9
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextWrapped = true

    task.defer(function()
        if logContainer and logContainer.Parent then
            logContainer.CanvasPosition = Vector2.new(0, logContainer.AbsoluteCanvasSize.Y)
        end
    end)

    task.delay(45, function()
        if label and label.Parent then
            label:Destroy()
        end
    end)
end

env.AddLog = AddLog

local function loadRuntimeState()
    if stateBusy then
        task.wait(0.3)
    end

    local state = cloneDefaultState()
    if type(isfile) == "function" and isfile(VISITED_SERVERS_FILE) then
        local ok, decoded = pcall(function()
            local content = readfile(VISITED_SERVERS_FILE)
            if not content or content == "" then
                return nil
            end
            return HttpService:JSONDecode(content)
        end)

        if ok and type(decoded) == "table" then
            state = decoded
            state.servers = type(state.servers) == "table" and state.servers or {}
            state.serverVisits = type(state.serverVisits) == "table" and state.serverVisits or {}
            state.serverBlockedUntil = type(state.serverBlockedUntil) == "table" and state.serverBlockedUntil or {}
            state.serverQueue = type(state.serverQueue) == "table" and state.serverQueue or {}
            state.serverQueueBuiltAt = tonumber(state.serverQueueBuiltAt) or 0
            state.serverQueueStrategy = tostring(state.serverQueueStrategy or "")
            state.serverAgeRawObservedAt = tonumber(state.serverAgeRawObservedAt) or 0
            state.preSpawnTargetCooldowns = type(state.preSpawnTargetCooldowns) == "table" and state.preSpawnTargetCooldowns or {}
            state.hops = tonumber(state.hops) or 0
            state.fruits = tonumber(state.fruits) or 0
            state.speed = tonumber(state.speed) or 230
        end
    end
    return state
end

local function saveRuntimeState(state)
    task.spawn(function()
        stateBusy = true
        pcall(function()
            if type(writefile) == "function" then
                writefile(VISITED_SERVERS_FILE, HttpService:JSONEncode(state))
            end
        end)
        stateBusy = false
    end)
end

end


-- DEV SERVER-AGE LEARNING ----------------------------------------------------
-- A JobId is unique to one live Roblox server instance. Once we land there,
-- TimeIn reveals its absolute start timestamp. Persisting that mapping lets us
-- recognize the same JobId later in __ServerBrowser without teleporting first.
local SERVER_AGE_CACHE_MAX_AGE = 36 * 60 * 60
local SERVER_AGE_CACHE_MAX_ENTRIES = 5000
local serverAgeCache = {version=1, servers={}}
local serverAgeCacheLoaded = false

local function loadServerAgeCache()
    if serverAgeCacheLoaded then return serverAgeCache end
    serverAgeCacheLoaded = true
    serverAgeCache = {version=1, servers={}}
    if type(isfile) == "function" and type(readfile) == "function" and isfile(SERVER_AGE_CACHE_FILE) then
        local ok, decoded = pcall(function()
            local raw = readfile(SERVER_AGE_CACHE_FILE)
            if type(raw) ~= "string" or raw == "" then return nil end
            return HttpService:JSONDecode(raw)
        end)
        if ok and type(decoded) == "table" then
            decoded.servers = type(decoded.servers) == "table" and decoded.servers or {}
            serverAgeCache = decoded
        end
    end

    local now = os.time()
    local kept = {}
    local count = 0
    for jobId, rec in pairs(serverAgeCache.servers or {}) do
        if type(rec) == "table" then
            local lastSeen = tonumber(rec.lastSeenAt) or 0
            local startedAt = tonumber(rec.startedAt) or 0
            if startedAt > 0 and lastSeen > 0 and (now - lastSeen) <= SERVER_AGE_CACHE_MAX_AGE then
                if rec.visitedByPrimeHub == nil then
                    rec.visitedByPrimeHub = true
                    rec.visitCount = math.max(1, tonumber(rec.visitCount) or 1)
                end
                kept[tostring(jobId)] = rec
                count += 1
            end
        end
    end
    serverAgeCache.servers = kept
    serverAgeCacheStats.entries = count
    return serverAgeCache
end

local function saveServerAgeCache()
    local cache = loadServerAgeCache()
    if type(writefile) ~= "function" then return false end
    return pcall(function()
        writefile(SERVER_AGE_CACHE_FILE, HttpService:JSONEncode(cache))
    end)
end

local function findWorldTimeIn()
    local origin = workspace:FindFirstChild("_WorldOrigin")
    local locations = origin and origin:FindFirstChild("Locations")
    if not locations then return nil, nil end

    local preferred = locations:FindFirstChild("Sea of Treats")
    if preferred then
        local v = preferred:GetAttribute("TimeIn")
        if type(v) == "number" and v > 0 then return v, preferred.Name end
    end

    for _, obj in ipairs(locations:GetChildren()) do
        local v = obj:GetAttribute("TimeIn")
        if type(v) == "number" and v > 0 then
            return v, obj.Name
        end
    end
    return nil, nil
end
