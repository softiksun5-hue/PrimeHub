        if not teamReady then
            AddLog("Fruit detected but character is not ready yet: " .. tostring(fruit.Name), Color3.fromRGB(255, 180, 80))

        local beforePickup = snapshotHeldFruitTools()
        local tool = nil

        local pickupOffset = isServerWorldFruitModel(fruit) and 2 or (tonumber(env.FlyOffset) or 5)
        local flightSpeed = math.max(50, tonumber(env.TweenSpeed) or 230)

        if firstHandle and firstRoot then
            local firstTarget = firstHandle.Position + Vector3.new(0, pickupOffset, 0)
            local firstDistance = (firstRoot.Position - firstTarget).Magnitude

                local expectedTravel = firstDistance / flightSpeed
                local flightTimeout = math.max(
                    20,
                    expectedTravel * (tonumber(env.FlightTimeoutMultiplier) or 1.35)
                        + (tonumber(env.FlightTimeoutBuffer) or 20)
                )
                executorLog("PICKUP", string.format(
                    "FLIGHT start distance=%.0f speed=%.0f expected=%.1fs timeout=%.1fs target=%s",
                    firstDistance, flightSpeed, expectedTravel, flightTimeout, tostring(fruit.Name)
                ))

                env.ViajandoAteAFruta = true
                local flightDeadline = tick() + flightTimeout
                local lastRemaining = firstDistance
                local bestRemaining = firstDistance

                local flightStalled = false

                    local root = getPlayerRoot()

                    local remaining = (root.Position - targetPosition).Magnitude
                    lastRemaining = remaining
                    if remaining <= 8 then break end

                    if remaining <= (bestRemaining - progressEpsilon) then
                        bestRemaining = remaining
                        lastProgressAt = tick()
                    elseif (tick() - lastProgressAt) >= noProgressTimeout then
                        flightStalled = true
                        break
                    end

                    if not applyFlight(targetPosition, flightSpeed) then break end
                    task.wait(0.1)
                end

                    executorLog("EVENT", string.format(

                    ))
                    return false
                end

                if flightStalled and lastRemaining > 20 then
                    executorLog("PICKUP", string.format(

                        noProgressTimeout, lastRemaining
                    ))
                    return false
                end

                local endHandle = getDroppedFruitPart(fruit)

                    local remaining = (endRoot.Position - (endHandle.Position + Vector3.new(0, pickupOffset, 0))).Magnitude
                    executorLog("PICKUP", string.format("FLIGHT end remaining=%.1f target=%s", remaining, tostring(fruit.Name)))
                end
            end
        end

                    root.CFrame = handle.CFrame + Vector3.new(0, pickupOffset, 0)
                    root.Velocity = Vector3.zero
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.RotVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                else
                    applyFlight(targetPosition, flightSpeed)
                end
            elseif not root then
                AddLog("Flight/pickup failed: root unavailable.", Color3.fromRGB(255, 100, 100))
                break
            end

            task.wait(0.1)
            tool = findPickedUpTool(beforePickup, preferredName, expectedOriginal)
        end
        cleanupFlight()

        if not tool then
            executorLog("PICKUP", string.format(
                "strict pickup failed for name=%q OriginalName=%q; unrelated inventory Tools ignored",
                tostring(preferredName), tostring(expectedOriginal)
            ))
            failedPickupTargets[fruit] = tick() + 30
            AddLog("Pickup timed out: " .. tostring(fruit.Name) .. " — blacklisting target and hopping.",
                Color3.fromRGB(255, 180, 80))
            executorLog("PICKUP", "failed WORLD_DROP blacklisted for 30s; AutoHop may leave current server")
            return false
        end

        if not toolMatchesFruitIdentity(tool, preferredName, expectedOriginal) then
            executorLog("WARN", "Rejected mismatched pickup Tool: " .. tostring(tool.Name))

            return false

        executorLog("STORE", "pickup detected ptr=" .. tostring(tool) .. " name=" .. tostring(tool.Name)
            .. " parent=" .. toolParentName(tool) .. " expectedOriginal=" .. tostring(expectedOriginal))
        task.wait(tonumber(env.StorePostPickupDelay) or 0.35)

        local refreshedTool, replacedAfterPickup = reacquireHeldFruitTool(
            tool, preferredName, beforePickup, env.StoreReacquireTimeout, expectedOriginal
        )
        if refreshedTool then
            if replacedAfterPickup then
                executorLog("STORE", "post-pickup Tool replacement detected: old=" .. tostring(tool)
                    .. " new=" .. tostring(refreshedTool))
            end
            tool = refreshedTool
        else
            dumpHeldFruitTools("post-pickup reacquire failed")
            AddLog("Could not reacquire the picked fruit; continuing hop cycle.", Color3.fromRGB(255, 180, 80))
            return false
        end

            tool, preferredName, beforePickup, expectedOriginal
        )
        task.wait(0.5)

        if stored then
            AddLog("Stored: " .. tostring(tool.Name), Color3.fromRGB(0, 255, 100))
            env.HoldingHighTierFruit = false
            return true
        end

            return handleHighTierFruit(tool, metadata)
