    return state
end

local PRIMEHUB_ART_FILE = "PrimeHub_cat-girl.png"
local PRIMEHUB_ART_URL = tostring(env.PrimeHubArtUrl or "https://raw.githubusercontent.com/softiksun5-hue/PrimeHub/main/assets/cat-girl.b64")
-- PrimeHub 8.0 omits the large embedded PNG from the teleport runtime.
-- UI reuses PrimeHub_cat-girl.png if present or fetches PRIMEHUB_ART_URL asynchronously.
local PRIMEHUB_ART_EMBEDDED_B64 = "" -- stripped to keep queued payload small
local function decodePrimeHubBase64(data)
    -- Prefer the executor's native decoder when available.
    local candidates = {}
    pcall(function()
        if type(crypt) == "table" then
            if type(crypt.base64decode) == "function" then candidates[#candidates+1] = crypt.base64decode end
            if type(crypt.base64) == "table" and type(crypt.base64.decode) == "function" then candidates[#candidates+1] = crypt.base64.decode end
        end
    end)
    pcall(function()
        if type(syn) == "table" and type(syn.crypt) == "table" and type(syn.crypt.base64) == "table" and type(syn.crypt.base64.decode) == "function" then
            candidates[#candidates+1] = syn.crypt.base64.decode
        end
    end)
    for _, fn in ipairs(candidates) do
        local ok, decoded = pcall(fn, data)
        if ok and type(decoded) == "string" and #decoded > 1000 then return decoded end
    end

    -- Pure-Lua fallback for executors without a crypto helper.
    local alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    local map = {}
    for i = 1, #alphabet do map[string.byte(alphabet, i)] = i - 1 end
    local out, n = {}, 0
    local i, len = 1, #data
    while i <= len do
        local c1, c2, c3, c4 = string.byte(data, i, i + 3)
        i = i + 4
        local a, b = map[c1], map[c2]
        if a and b then
            local cc = c3 == 61 and nil or map[c3]
            local d = c4 == 61 and nil or map[c4]
            n=n+1; out[n]=string.char(a * 4 + math.floor(b / 16))
            if cc then
                n=n+1; out[n]=string.char((b % 16) * 16 + math.floor(cc / 4))
                if d then
                    n=n+1; out[n]=string.char((cc % 4) * 64 + d)
                end
            end
        end
    end
    return table.concat(out)
end

local function ensurePrimeHubArtAsset()
    local asset = nil
    local function registerExisting()
        if type(isfile) ~= "function" or not isfile(PRIMEHUB_ART_FILE) then return nil end
        local registered = nil
        if type(getcustomasset) == "function" then
            local ok, value = pcall(getcustomasset, PRIMEHUB_ART_FILE)
            if ok then registered = value end
        elseif type(getsynasset) == "function" then
            local ok, value = pcall(getsynasset, PRIMEHUB_ART_FILE)
            if ok then registered = value end
        end
        return registered
    end

    asset = registerExisting()
    if asset then return asset end

    pcall(function()
        if type(writefile) == "function" then
            local png = decodePrimeHubBase64(PRIMEHUB_ART_EMBEDDED_B64)
            if type(png) == "string" and #png > 1000 then writefile(PRIMEHUB_ART_FILE, png) end
        end
    end)
    asset = registerExisting()
    return asset
end
