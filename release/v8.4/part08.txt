             20,
                    expectedTravel * (tonumber(env.FlightTimeoutMultiplier) or 1.35)
                        + (tonumber(env.FlightTimeoutBuffer) or 20)
                )
                executorLog("PICKUP", string.format(
                    "FLIGHT start distance=%.0f speed=%.0f expected=%.1fs timeout=%.1fs target=%s",
                    firstDistance, flightSpeed, expectedTravel, flightTimeout, tostring(fruit.Name)
                ))

                env.ViajandoAteAFruta = true
                local flightDeadline = tick() + flightTimeout
                local lastRemaining = firstDistance
                local bestRemaining = firstDistance
                local lastProgressAt = tick()
                local flightStalled = false
                local progressEpsilon = tonumber(env.FlightProgressEpsilon) or 75
                local noProgressTimeout = tonumber(env.FlightNoProgressTimeout) or 15
                local targetLeftWorkspace = false
                local targetStructureLost = false
                while runtimeActive() and tick() < flightDeadline do
                    local inWorkspace = false
                    pcall(function() inWorkspace = fruit:IsDescendantOf(workspace) end)
                    if not fruit.Parent or not inWorkspace then
                        targetLeftWorkspace = true
                        break
                    end
                    if not isServerWorldFruitModel(fruit) then
                        targetStructureLost = true
                        break
                    end

                    local handle = getDroppedFruitPart(fruit)
                    local root = getPlayerRoot()
                    local character = LocalPlayer.Character
                    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                    if not handle or not root or not humanoid or humanoid.Health <= 0 then break end

                    local targetPosition = handle.Position + Vector3.new(0, pickupOffset, 0)
                    local remaining = (root.Position - targetPosition).Magnitude
                    lastRemaining = remaining
                    if remaining <= 8 then break end

                    if remaining <= (bestRemaining - progressEpsilon) then
                        bestRemaining = remaining
                        lastProgressAt = tick()
                    elseif (tick() - lastProgressAt) >= noProgressTimeout then
                        flightStalled = true
                        break
                    end

                    if not applyFlight(targetPosition, flightSpeed) then break end
                    task.wait(0.1)
                end
                cleanupFlight()

                if (targetLeftWorkspace or targetStructureLost) and lastRemaining > 20 then
                    executorLog("EVENT", string.format(
                        "WORLD_DROP became unavailable during flight remaining=%.0f leftWorkspace=%s structureLost=%s; aborting immediately",
                        lastRemaining, tostring(targetLeftWorkspace), tostring(targetStructureLost)
                    ))
                    return false
                end

                if flightStalled and lastRemaining > 20 then
                    executorLog("PICKUP", string.format(
                        "FLIGHT watchdog: no meaningful progress for %.1fs remaining=%.0f; aborting target",
                        noProgressTimeout, lastRemaining
                    ))
                    return false
                end

                local endRoot = getPlayerRoot()
                local endHandle = getDroppedFruitPart(fruit)
                if endRoot and endHandle then
                    local remaining = (endRoot.Position - (endHandle.Position + Vector3.new(0, pickupOffset, 0))).Magnitude
                    executorLog("PICKUP", string.format("FLIGHT end remaining=%.1f target=%s", remaining, tostring(fruit.Name)))
                end
            end
        end

        local deadline = tick() + 5.0

        while not tool and runtimeActive() and tick() < deadline do
            if not fruit.Parent then
                tool = findPickedUpTool(beforePickup, preferredName, expectedOriginal)
                if tool then break end
            end

            local handle = getDroppedFruitPart(fruit)
            local root = getPlayerRoot()
            if handle and root then
                local targetPosition = handle.Position + Vector3.new(0, pickupOffset, 0)
                local distance = (root.Position - targetPosition).Magnitude

                if distance <= 8 then
                    root.CFrame = handle.CFrame + Vector3.new(0, pickupOffset, 0)
                    root.Velocity = Vector3.zero
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.RotVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                else
                    applyFlight(targetPosition, flightSpeed)
                end
            elseif not root then
                AddLog("Flight/pickup failed: root unavailable.", Color3.fromRGB(255, 100, 100))
                break
            end

            task.wait(0.1)
            tool = findPickedUpTool(beforePickup, preferredName, expectedOriginal)
        end
        cleanupFlight()

        if not tool then
            executorLog("PICKUP", string.format(
                "strict pickup failed for name=%q OriginalName=%q; unrelated inventory Tools ignored",
                tostring(preferredName), tostring(expectedOriginal)
            ))
            failedPickupTargets[fruit] = tick() + 30
            AddLog("Pickup timed out: " .. tostring(fruit.Name) .. " — blacklisting target and hopping.",
                Color3.fromRGB(255, 180, 80))
            executorLog("PICKUP", "failed WORLD_DROP blacklisted for 30s; AutoHop may leave current server")
            return false
        end

        if not toolMatchesFruitIdentity(tool, preferredName, expectedOriginal) then
            executorLog("WARN", "Rejected mismatched pickup Tool: " .. tostring(tool.Name))
            AddLog("Pickup identity mismatch; ignoring " .. tostring(tool.Name), Color3.fromRGB(255, 180, 80))
            return false
        end

        executorLog("STORE", "PICKUP TOOL CONFIRMED ptr=" .. tostring(tool) .. " name=" .. tostring(tool.Name)
            .. " parent=" .. toolParentName(tool) .. " expectedOriginal=" .. tostring(expectedOriginal))

        -- Count a fruit as soon as pickup itself is positively confirmed.
        -- Storage is a separate best-effort step and must not decide whether
        -- the successful pickup appears in the FRUITS counter.
        pcall(updateStats, nil, fruitCount + 1)
        AddLog("Picked up: " .. tostring(tool.Name), Color3.fromRGB(0, 255, 100))
        executorLog("PICKUP", "PICKUP COUNTED; fruit counter incremented before storage")

        task.wait(tonumber(env.StorePostPickupDelay) or 0.35)

        local refreshedTool, replacedAfterPickup = reacquireHeldFruitTool(
            tool, preferredName, beforePickup, env.StoreReacquireTimeout, expectedOriginal
        )
        if refreshedTool then
            if replacedAfterPickup then
                executorLog("STORE", "post-pickup Tool replacement detected: old=" .. tostring(tool)
                    .. " new=" .. tostring(refreshedTool))
            end
            tool = refreshedTool
        else
            dumpHeldFruitTools("post-pickup reacquire failed")
            AddLog("Could not reacquire the picked fruit; continuing hop cycle.", Color3.fromRGB(255, 180, 80))
            return false
        end

        local stored, storeReason, storeDetail = storeFruit(
            tool, preferredName, beforePickup, expectedOriginal
        )
        task.wait(0.35)

        if stored then
            AddLog("Stored: " .. tostring(tool.Name), Color3.fromRGB(0, 255, 100))
            executorLog("STORE", "STORE CONFIRMED; pickup was already counted independently")
            env.HoldingHighTierFruit = false
            return true
        end

        local metadata = FruitMetadata[preferredName or tool.Name] or FruitMetadata[tool.Name]

        if storeReason == "full" then
            return handleHighTierFruit(tool, metadata)
        end

        AddLog(
            "Could not confirm storage (" .. tostring(storeReason) .. "): " .. tostring(storeDetail),
            Color3.fromRGB(255, 100, 100)
        )

        -- PrimeHub 7.9: an unconfirmed remote response must not permanently kill the
        -- server-hop loop. The strict identity checks above already prevent us from
        -- storing a completely unrelated inventory fruit.
        env.HoldingHighTierFruit = false
        AddLog("Storage attempt failed; AutoHop will continue.", Color3.fromRGB(255, 200, 0))
        return false
    end)

    processingFruit = false
    activeWorldDropTarget = nil
    if not ok then
        AddLog("processFruit error: " .. tostring(result), Color3.fromRGB(255, 100, 100))
        cleanupFlight()
        if not env.HoldingHighTierFruit then schedulePostFruitHop("process error") end
        return false
    end

    if not env.HoldingHighTierFruit then
        schedulePostFruitHop(result == true and "fruit resolved" or "fruit attempt finished")
    end
    return result == true
end

local seenTeleportPrompts = setmetatable({}, {__mode = "k"})

local function collectPromptText(prompt)
    local collected = ""
    if not prompt then return collected end
    local descendants = {}
    pcall(function() descendants = prompt:GetDescendants() end)
    for _, node in ipairs(descendants) do
        local isText = false
        pcall(function()
            isText = node:IsA("TextLabel") or node:IsA("TextButton")
        end)
        if isText and type(node.Text) == "string" then
            collected = collected .. " " .. node.Text
        end
    end
    return collected
end

local function observeTeleportErrorPrompt()
    -- IMPORTANT: Stage 11 destroyed the prompt instance itself. Roblox kept the
    -- modal dimmer, leaving the whole screen grey. PrimeHub 7.9 only observes/logs.
    local promptGui = nil
    pcall(function() promptGui = CoreGui:FindFirstChild("RobloxPromptGui") end)
    local overlay = nil
    pcall(function() overlay = promptGui and promptGui:FindFirstChild("promptOverlay") end)
    if not overlay then return nil end

    local children = {}
    pcall(function() children = overlay:GetChildren() end)
    for _, prompt in ipairs(children) do
        local text = collectPromptText(prompt)
        local lower = string.lower(text)
        local isHopFailure = string.find(lower, "260", 1, true)
            or string.find(lower, "277", 1, true)
            or string.find(lower, "279", 1, true)
            or string.find(lower, "773", 1, true)
            or string.find(lower, "server is full", 1, true)
            or string.find(lower, "server full", 1, true)
            or string.find(lower, "receiving data", 1, true)
            or string.find(lower, "lost connection", 1, true)
            or string.find(lower, "disconnected", 1, true)
            or string.find(lower, "failed to connect", 1, true)
            or string.find(lower, "teleport", 1, true)
            or string.find(lower, "unauthorized", 1, true)

        if isHopFailure and text ~= "" then
            local previous = seenTeleportPrompts[prompt]
            if previous ~= text then
                seenTeleportPrompts[prompt] = text
                hopDebug("Roblox teleport prompt observed: " .. text)
                return text
            end
        end
    end
    return nil
end

local function isConnectionClassFailure(reason)
    local lower = string.lower(tostring(reason or ""))
    return string.find(lower, "260", 1, true) ~= nil
        or string.find(lower, "277", 1, true) ~= nil
        or string.find(lower, "279", 1, true) ~= nil
        or string.find(lower, "receiving data", 1, true) ~= nil
        or string.find(lower, "lost connection", 1, true) ~= nil
        or string.find(lower, "disconnected", 1, true) ~= nil
        or string.find(lower, "failed to connect", 1, true) ~= nil
end
