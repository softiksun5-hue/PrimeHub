-- Lightweight current-server timing only; no prespawn/postspawn targeting.
local function currentServerTiming()
    local startedAt, sourceName = nil, nil
    pcall(function()
        local origin = workspace:FindFirstChild("_WorldOrigin")
        local locations = origin and origin:FindFirstChild("Locations")
        if locations then
            for _, obj in ipairs(locations:GetChildren()) do
                local value = obj:GetAttribute("TimeIn")
                if type(value) == "number" and value > 0 then
                    startedAt, sourceName = value, tostring(obj.Name)
                    break
                end
            end
        end
    end)
    if not startedAt then return nil end
    local okNow, serverNow = pcall(function() return workspace:GetServerTimeNow() end)
    if not okNow or type(serverNow) ~= "number" then return nil end
    local age = math.max(0, serverNow - startedAt)
    return {
        startedAt=startedAt, age=age, phase=age % 3600,
        untilHourly=(3600 - (age % 3600)) % 3600,
        source=sourceName or "TimeIn",
    }
end
