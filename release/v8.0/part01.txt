local function executorLog(level, message)
    if not env.PrimeHubDebug then return end
    local stamp = os.date("%H:%M:%S")
    local line = string.format("[PRIMEHUB][%s][%s] %s", stamp, tostring(level or "INFO"), tostring(message))
    if level == "ERROR" or level == "WARN" then
        local ok = pcall(function() warn(line) end)
        if not ok then pcall(function() print(line) end) end
    else
        pcall(function() print(line) end)
    end
end
