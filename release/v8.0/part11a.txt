         n=n+1; out[n]=string.char(a * 4 + math.floor(b / 16))
            if cc then
                n=n+1; out[n]=string.char((b % 16) * 16 + math.floor(cc / 4))
                if d then
                    n=n+1; out[n]=string.char((cc % 4) * 64 + d)
                end
            end
        end
    end
    return table.concat(out)
end

local function ensurePrimeHubArtAsset()
    local asset = nil
    local function registerExisting()
        if type(isfile) ~= "function" or not isfile(PRIMEHUB_ART_FILE) then return nil end
        local registered = nil
        if type(getcustomasset) == "function" then
            local ok, value = pcall(getcustomasset, PRIMEHUB_ART_FILE)
            if ok then registered = value end
        elseif type(getsynasset) == "function" then
            local ok, value = pcall(getsynasset, PRIMEHUB_ART_FILE)
            if ok then registered = value end
        end
        return registered
    end

    asset = registerExisting()
    if asset then return asset end

    pcall(function()
        if type(writefile) == "function" then
            local png = decodePrimeHubBase64(PRIMEHUB_ART_EMBEDDED_B64)
            if type(png) == "string" and #png > 1000 then writefile(PRIMEHUB_ART_FILE, png) end
        end
    end)
    asset = registerExisting()
    return asset
end
