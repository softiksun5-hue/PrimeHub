env.HoldingHighTierFruit do
            local fruits, hopStats = scanWorkspaceForFruits()
            if #fruits > 0 then
                fruitDetected = true
                AddLog("Fruit appeared before hop; switching to sniper.", Color3.fromRGB(0, 255, 100))
                ensureTeamReady("fruit appeared before hop")
                return
            end

            if hopStats and (tonumber(hopStats.pendingWorldDrops) or 0) > 0 then
                if statusLabel then statusLabel.Text = "WORLD_DROP • WAIT" end
                task.wait(math.min(
                    0.35,
                    math.max(0.12, tonumber(hopStats.pendingConfirmRemaining) or 0.25)
                ))
            else
                if ServerHop() then return end
                if statusLabel then statusLabel.Text = "HOP RETRY..." end
                task.wait(math.max(1, tonumber(env.HopNoCandidateDelay) or 2))
            end
        end
    end

    if statusLabel then statusLabel.Text = "RADAR ACTIVE" end
end

local function periodicSniperWorker()
    while runtimeActive() do
        task.wait(math.max(0.5, tonumber(env.ScanInterval) or 3))
        if not runtimeActive() then break end

        local fruits = scanWorkspaceForFruits()
        fruitDetected = #fruits > 0

        if env.AutoFruitSniper and fruitDetected and not env.HoldingHighTierFruit then
            AddLog("Found " .. tostring(#fruits) .. " target fruit(s)", Color3.fromRGB(0, 255, 100))
            local teamReady = ensureTeamReady("periodic scan found fruit")
            if not teamReady then
                AddLog("Waiting for team/character before fruit pickup.", Color3.fromRGB(255, 180, 80))
                task.wait(0.75)
            end
            for _, fruit in ipairs(fruits) do
                if not runtimeActive() or not env.AutoFruitSniper or env.HoldingHighTierFruit then break end
                processFruit(fruit)
                task.wait(0.5)
            end
        end
    end
end


-- 8.0 diagnostics: track real WORLD_DROP lifetime.
local worldDropLifecycle = setmetatable({}, {__mode="k"})

local function worldDropDistance(instance)
    local part = getDroppedFruitPart(instance)
    local root = getPlayerRoot()
    if part and root then
        return (part.Position - root.Position).Magnitude
    end
    return -1
end

local function monitorWorldDropLifecycle(instance, origin)
    if not instance or worldDropLifecycle[instance] then return end

    local record = {
        seenAt = tick(),
        origin = tostring(origin or "unknown"),
        startDistance = worldDropDistance(instance),
    }
    worldDropLifecycle[instance] = record

    local path = tostring(instance)
    pcall(function() path = instance:GetFullName() end)
    executorLog("EVENT", string.format(
        "WORLD_DROP track start origin=%s distance=%.0f path=%s",
        record.origin, record.startDistance, path
    ))

    local connection
    local childConnection
    childConnection = instance.ChildRemoved:Connect(function(child)
        local childName = tostring(child and child.Name or "nil")
        if childName == "Handle" or childName == "Fruit" or childName == "FruitAnimator" then
            local lifetime = math.max(0, tick() - record.seenAt)
            executorLog("EVENT", string.format(
                "WORLD_DROP structure removed child=%s lifetime=%.2fs activeTarget=%s recoveryGrace=%.2fs",
                childName, lifetime, tostring(activeWorldDropTarget == instance),
                tonumber(env.WorldDropRecoveryGrace) or 3.0
            ))
        end
    end)

    connection = instance.AncestryChanged:Connect(function()
        task.defer(function()
            if not worldDropLifecycle[instance] then
                if connection then connection:Disconnect() end
                return
            end

            local inWorkspace = false
            pcall(function() inWorkspace = instance:IsDescendantOf(workspace) end)
            if inWorkspace then return end

            local lifetime = math.max(0, tick() - record.seenAt)
            local endDistance = worldDropDistance(instance)
            local active = activeWorldDropTarget == instance
            local parentName = "nil"
            pcall(function()
                if instance.Parent then parentName = instance.Parent:GetFullName() end
            end)

            executorLog("EVENT", string.format(
                "WORLD_DROP track end origin=%s lifetime=%.2fs startDistance=%.0f endDistance=%.0f activeTarget=%s parent=%s",
                record.origin, lifetime, record.startDistance, endDistance, tostring(active), parentName
            ))

            worldDropLifecycle[instance] = nil
            if connection then connection:Disconnect() end
            if childConnection then childConnection:Disconnect() end
        end)
    end)
end

local function worldDropEventWorker()
    local eventCount = 0
    local signal = CollectionService:GetInstanceAddedSignal("WORLD_DROP")
    local connection

    local function handleWorldDrop(instance)
        monitorWorldDropLifecycle(instance, "tag-added")
        eventCount = eventCount + 1
        local eventId = eventCount
        local observedAt = tick()
        task.spawn(function()
            local ready = false
            local attempts = 0
            local confirmDeadline = tick() + math.max(1.5, tonumber(env.WorldDropRuntimeConfirm) or 0.65) + 1.0
            while runtimeActive() and tick() < confirmDeadline do
                attempts = attempts + 1
                if not instance or not instance.Parent then return end
                local confirmed, pending = worldDropConfirmStatus(instance)
                if confirmed then
                    ready = true
                    break
                end
                if not pending and attempts > 3 then
                    break
                end
                task.wait(0.1)
            end

            local path = tostring(instance)
            pcall(function() path = instance:GetFullName() end)
            local distance = -1
            local part = getDroppedFruitPart(instance)
            local root = getPlayerRoot()
            if part and root then distance = (part.Position - root.Position).Magnitude end

            executorLog("EVENT", string.format(
                "WORLD_DROP #%d ready=%s attempts=%d readyIn=%.2fs distance=%.0f path=%s",
                eventId, tostring(ready), attempts, tick() - observedAt, distance, path
            ))

            if ready and env.AutoFruitSniper and not env.HoldingHighTierFruit and not processingFruit then
                fruitDetected = true
                AddLog("[PrimeHub] WORLD_DROP event detected. Starting sniper immediately.", Color3.fromRGB(0, 255, 100))
                processFruit(instance)
            end
        end)
    end

    connection = signal:Connect(handleWorldDrop)
    local existingCount = 0
    local okExisting, existing = pcall(function() return CollectionService:GetTagged("WORLD_DROP") end)
    if okExisting and type(existing) == "table" then
        existingCount = #existing
        for _, instance in ipairs(existing) do
            monitorWorldDropLifecycle(instance, "startup-existing")
        end
    end
    executorLog("EVENT", "WORLD_DROP listener active existingTagged=" .. tostring(existingCount))

    while runtimeActive() do task.wait(1) end
    if connection then connection:Disconnect() end
end

local function autoHopWorker()
    local elapsed = 0
    while runtimeActive() do
        task.wait(1)
        if not runtimeActive() then break end

        if env.AutoHop and not env.HoldingHighTierFruit and not hopping then
            local fruits, autoHopStats = scanWorkspaceForFruits()
            fruitDetected = #fruits > 0
            local worldDropPending = autoHopStats
                and (tonumber(autoHopStats.pendingWorldDrops) or 0) > 0

            if fruitDetected or processingFruit or worldDropPending then
                elapsed = 0
            else
                local now = tick()

                if postFruitHopDeadline ~= nil then
                    if now >= postFruitHopDeadline then
                        postFruitHopDeadline = nil
                        AddLog("Fast hop triggered", Color3.fromRGB(0, 200, 255))
                        if ServerHop() then return end
                        local retryDelay = math.max(1, tonumber(env.HopNoCandidateDelay) or 2)
                        postFruitHopDeadline = tick() + retryDelay
                        executorLog("HOP", "fast retry scheduled in " .. tostring(retryDelay) .. "s")
                        elapsed = 0
                    end
                else
                    elapsed = elapsed + 1
                    local interval = math.max(5, tonumber(env.AutoHopInterval) or 30)
                    if elapsed >= interval then
                        AddLog("Auto-hop timer triggered (" .. tostring(interval) .. "s)", Color3.fromRGB(0, 200, 255))
                        if ServerHop() then return end
                        local retryDelay = math.max(1, tonumber(env.HopNoCandidateDelay) or 2)
                        postFruitHopDeadline = tick() + retryDelay
                        executorLog("HOP", "fast retry scheduled in " .. tostring(retryDelay) .. "s")
                        elapsed = 0
                    end
                end
            end
        else
            elapsed = 0
        end
    end
end

        if subStatus then
            subStatus.Text = string.format(
                "Server: %s | Hops: %d | Fruits: %d",
                shortJobId(), hopCount, fruitCount
            )
        end
    end
end

local function getESPHandle(child)
    return getDroppedFruitPart(child)
end

local function destroyFruitESP()
    for _, child in ipairs(workspace:GetChildren()) do
        local handle = getDroppedFruitPart(child)
        local marker = handle and handle:FindFirstChild("PrimeHub_ESP")
        if marker then pcall(function() marker:Destroy() end) end
    end
end

local function createFruitESP(fruit)
    local handle = getESPHandle(fruit)
    if not handle or handle:FindFirstChild("PrimeHub_ESP") then return end

    local folder = Instance.new("Folder", handle)
    folder.Name = "PrimeHub_ESP"

    local billboard = Instance.new("BillboardGui", folder)
    billboard.Name = "FruitBillboard"
    billboard.Adornee = handle
    billboard.Size = UDim2.new(0, 140, 0, 40)
    billboard.StudsOffset = Vector3.new(0, 2, 0)
    billboard.AlwaysOnTop = true

    local label = Instance.new("TextLabel", billboard)
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.fromRGB(255, 80, 80)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 11
    label.TextStrokeTransparency = 0.4

    task.spawn(function()
        while runtimeActive() and env.FruitESP and label.Parent and handle.Parent do
            local root = getPlayerRoot()
            if root then
                local distance = math.floor((root.Position - handle.Position).Magnitude)
                local displayName = isServerWorldFruitModel(fruit) and "SERVER FRUIT" or tostring(fruit.Name)
                label.Text = displayName .. " (" .. tostring(distance) .. "m)"
            end
            task.wait(1)
        end
        if folder and folder.Parent then pcall(function() folder:Destroy() end) end
    end)
end

local function fruitESPWorker()
    local wasEnabled = false
    while runtimeActive() do
        if env.FruitESP then
            wasEnabled = true
            for _, child in ipairs(scanWorkspaceForFruits()) do
                pcall(function() createFruitESP(child) end)
            end
        elseif wasEnabled then
            destroyFruitESP()
            wasEnabled = false
        end
        task.wait(1)
    end
    destroyFruitESP()
end

local function startWorkers()
    executorLog("PERF", string.format("workers starting bootElapsed=%.2fs", tick() - PRIMEHUB_BOOT_TICK))
    local __selfBytes = type(env.PrimeHubSelfSource) == "string" and #env.PrimeHubSelfSource or -1
    executorLog("PERF", "self-contained runtime source bytes=" .. tostring(__selfBytes))
    if __resumeHopTransitSeconds then
        executorLog("PERF", "teleport-to-runtime transit=" .. tostring(__resumeHopTransitSeconds) .. "s")
    end
    task.spawn(worldDropEventWorker)
    task.spawn(fruitESPWorker)
    task.spawn(deepScanWorker)
    task.spawn(autoHopWorker)
    task.spawn(periodicSniperWorker)
    task.spawn(statusMonitorWorker)
end

end

            candidates[#candidates+1] = syn.crypt.base64.decode
        end
