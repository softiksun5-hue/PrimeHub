env.PrimeHubStoreHeldFruit = function()
    dumpHeldFruitTools("manual store requested")
    local tool = findFruitTool(nil, nil)
    if not tool then
        AddLog("Manual store: no held fruit Tool found.", Color3.fromRGB(255, 180, 80))
        return false, "no_tool"
    end

    AddLog("Manual store requested for " .. tostring(tool.Name), Color3.fromRGB(120, 200, 255))
    local stored, reason, detail = storeFruit(tool, tool.Name, nil)
    if stored then
        env.HoldingHighTierFruit = false
        AddLog("Manual store success: " .. tostring(tool.Name), Color3.fromRGB(0, 255, 100))
    else
        AddLog("Manual store failed: " .. tostring(reason) .. " / " .. tostring(detail), Color3.fromRGB(255, 100, 100))
    end
    return stored, reason, detail
end

env.PrimeHubServerQueueStatus = function()
    local state = loadRuntimeState()
    local queue = type(state.serverQueue) == "table" and state.serverQueue or {}
    local highCount, midCount, lowCount = 0, 0, 0
    for _, server in ipairs(queue) do
        local playing = tonumber(server.playing) or 0
        if playing >= 8 then highCount=highCount+1
        elseif playing >= 5 then midCount=midCount+1
        else lowCount=lowCount+1 end
    end
    local info = {
        queued=#queue, highQueued=highCount, midQueued=midCount, lowQueued=lowCount,
        strategy=tostring(state.serverQueueStrategy or ""), target=env.HopQueueTarget,
        builtAt=tonumber(state.serverQueueBuiltAt) or 0, minPlayers=env.HopQueueMinPlayers,
        maxPlayers=env.HopQueueMaxPlayers, cooldown=env.HopVisitedCooldown,
        botServerCooldown=env.BotServerCooldown, botServerScreening=env.BotServerScreening,
    }
    hopDebug(string.format(
        "QUEUE STATUS strategy=%s queued=%d high=%d mid=%d low=%d target=%d eligibleBand=%d-%d cooldown=%ss botCooldown=%ss builtAt=%s",
        info.strategy, info.queued, info.highQueued, info.midQueued, info.lowQueued, tonumber(info.target) or 0,
        tonumber(info.minPlayers) or 0, tonumber(info.maxPlayers) or 0, tostring(info.cooldown),
        tostring(info.botServerCooldown), tostring(info.builtAt)
    ))
    return info
end
